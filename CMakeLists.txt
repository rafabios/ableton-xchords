cmake_minimum_required(VERSION 3.22)
project(XChords VERSION 0.5.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
include(CTest)
add_library(xchords_core src/Theory.cpp)
target_include_directories(xchords_core PUBLIC src)
add_executable(xchords_tests tests/TheoryTests.cpp)
target_link_libraries(xchords_tests PRIVATE xchords_core)
add_test(NAME theory COMMAND xchords_tests)
add_executable(xchords_instrument_tests tests/InstrumentTests.cpp)
target_link_libraries(xchords_instrument_tests PRIVATE xchords_core)
add_test(NAME instruments COMMAND xchords_instrument_tests)
option(XCHORDS_BUILD_PLUGIN "Build standalone and VST3" ON)
if(XCHORDS_BUILD_PLUGIN)
  include(FetchContent)
  FetchContent_Declare(JUCE URL https://github.com/juce-framework/JUCE/archive/3fb5cbc0340758596dac39f7c8de47d562b4896e.tar.gz URL_HASH SHA256=2bb7095497a7bb7269dbdf841a7c6aa0939eaeb2a2bc9724ac4758d3ae3565a7 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(JUCE)
  juce_add_binary_data(XChordsPresets HEADER_NAME FactoryData.h NAMESPACE FactoryData SOURCES resources/factory-presets.xml)
  juce_add_plugin(XChords COMPANY_NAME "Vemcompy Labs" BUNDLE_ID "com.xchords.plugin" PLUGIN_MANUFACTURER_CODE Xchd PLUGIN_CODE Xc01 PRODUCT_NAME "Xchords - Harmony" FORMATS Standalone VST3 IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(XChords PRIVATE src/Plugin.cpp)
  target_link_libraries(XChords PRIVATE xchords_core juce::juce_audio_utils PUBLIC XChordsPresets juce::juce_recommended_config_flags)
  target_compile_definitions(XChords PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  if(BUILD_TESTING)
    juce_add_console_app(XChordsSmoke PRODUCT_NAME "XChordsSmoke")
    target_sources(XChordsSmoke PRIVATE tests/PluginTests.cpp tests/FeatureTests.cpp tests/InstrumentUiTests.cpp tests/FactoryPresetTests.cpp)
    target_include_directories(XChordsSmoke PRIVATE src)
    target_compile_definitions(XChordsSmoke PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1)
    target_link_libraries(XChordsSmoke PRIVATE XChords xchords_core juce::juce_audio_utils)
    add_test(NAME plugin_midi COMMAND XChordsSmoke)
  endif()
endif()
